# Names for stock lk (G3D1 V1.08) code and data, by offset in the lk payload.
# Each name becomes a linker alias that src/ can call or use.
# Calling conventions were confirmed on the device with the 2026-11-24 key test.
01100c lk_time_us            # u64 (void): microseconds since boot
06c0b4 lk_delay_us           # void (u32 us): busy wait
020530 lk_detect_key         # int (int key): bit 0 set while held; 0 recovery key, 1 volume up, 2 power
043698 lk_console_printf     # void (const char *fmt, ...): text console under the logo
043780 lk_console_goto       # void (int row, int col)
0437b0 lk_console_rows       # int (void): console rows
0437bc lk_console_clear      # void (void): fill the frame buffer with black and show it
043b70 lk_wdt_kick           # void (void): restart the hardware watchdog (TOPRGU +0x08 = 0x1971);
                             #   the stock lk sets it to 30 s (f_0037f8). Console text shows by itself.
0c9918 lk_boot_mode          # u8: boot mode (0 normal, 8 power-off charging)
05b948 lk_orange_warning     # the unlocked (orange state) warning screen
01a810 lk_get_slot_suffix    # const char *(void): "_a" or "_b", the slot this lk booted
01ac70 lk_set_active_slot    # int (const char *suffix): "_a"/"_b" -> misc boot_ctrl; 0 on success
01cc24 lk_platform_halt      # void (int action, int reason): action 1 reboot, 2 power off;
                             #   reason 1 PMIC cold reset, 7 watchdog reset (keeps the RGU boot mode)
043a74 lk_rgu_set_boot_mode  # void (u8 mode): TOPRGU +0x24 bits 0-3 for the next boot:
                             #   2 recovery (with a misc message), 3 bootloader (fastboot)
06b310 lk_bio_open           # void *(const char *name): block device by partition name
06b39c lk_bio_read           # long (void *dev, void *buf, long offset, unsigned long len)
06b40c lk_bio_write          # long (void *dev, const void *buf, long offset, unsigned long len)
06b2a0 lk_bio_close          # void (void *dev)
06c68c lk_malloc             # void *(unsigned long size)
06c7c8 lk_free               # void (void *ptr)
06dcec memset                # void *(void *s, int c, unsigned long n)
06f520 lk_dprintf            # int (const char *fmt, ...): LK log, not the screen
